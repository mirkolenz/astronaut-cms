import { Tabs, TabItem } from "@astrojs/starlight/components";
import Steps from "@/components/Steps.astro";
import StepItem from "@/components/StepItem.astro";

<Steps>
  <StepItem title="Install the package">
    <Tabs>
      <TabItem label="npm">
        ```sh
        npm install astronaut-cms @astrojs/react react react-dom zod
        ```
      </TabItem>
      <TabItem label="pnpm">
        ```sh
        pnpm add astronaut-cms @astrojs/react react react-dom zod
        ```
      </TabItem>
    </Tabs>
  </StepItem>
  <StepItem title="Add the integration to your Astro config">
    ```ts title="astro.config.ts"
    import { defineConfig } from "astro/config";
    import mdx from "@astrojs/mdx";
    import react from "@astrojs/react";
    import cms from "astronaut-cms";

    export default defineConfig({
      integrations: [mdx(), react(), cms()],
    });
    ```

  </StepItem>
  <StepItem title="Define your content collections">
    Create a `src/content.config.ts` file using Astro's standard content collection API:

    ```ts title="src/content.config.ts"
    import { defineCollection } from "astro:content";
    import { glob } from "astro/loaders";
    import { z } from "astro/zod";

    const blog = defineCollection({
      loader: glob({ pattern: "**/*.{md,mdx}", base: "src/content/blog" }),
      schema: ({ image }) =>
        z.object({
          title: z.string(),
          description: z.string(),
          date: z.coerce.date(),
          draft: z.boolean().default(false),
          cover: image().optional(),
        }),
    });

    export const collections = { blog };
    ```

  </StepItem>
  <StepItem title="Start the dev server and open the admin panel">
    ```sh
    npm run dev
    ```

    Navigate to `http://localhost:4321/admin` to access the CMS admin panel.

  </StepItem>
</Steps>

## Deploying

`astro dev` edits the working tree and needs no sign-in, so everything above works with no further setup.
A deployed CMS is different: it holds the repository session on the server, so it needs two things.

<Steps>
  <StepItem title="An SSR adapter">
    `@astrojs/node`, `@astrojs/cloudflare`, `@astrojs/vercel`, `@astrojs/netlify` — whichever suits
    your host. The rest of your site can stay static. The build fails with a message if one is
    missing, unless `admin: false` leaves the admin out of the build.
  </StepItem>

  <StepItem title="A Git backend">
    A [GitHub App or GitLab application](../guides/storage-backends/), whose client id goes in
    `astro.config.ts` and whose secret goes in the environment as
    `ASTRONAUT_CMS_GITHUB_CLIENT_SECRET` (or `..._GITLAB_...`).

    That is the only secret. The session cookie's key is derived from it, so there is nothing to
    generate, seed or keep in sync.

  </StepItem>
</Steps>
